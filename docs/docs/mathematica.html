(*
This package processes sparse sets and multisets through user-defined circuits.
Data flow is controlled by paths that connect various circuit components.
Associative memory functionality is isolated in specific types
of memory components.

Paths are either "digital" sets of integers i > 0, or "analog"
multisets of integers i > 0 for excitatory and i < 0 for inhibitory 
signals.

Partitioned multisets may be formed through the disjoint aggregation
of individual multisets. We refer to this mechanism as block coding.

All components must properly handle the presence of negative elements. 
The memory components systematically apply and then drop inhibitory 
elements, because the associative memory algorithms are inherently digital.
*)


(* 
Summates excitatory and inhibitory signals.
Outputs a net multiset. 
No element will exist as both positive and negative. 
*)

Multiset[u___List] := Module[{x, tally},

    x = Flatten[{u}];
    
    (* Tally positives, tally negatives, and add them together *)
    tally = Merge[{
                Counts[Select[Positive][x]], 
                (* Shift to positive, but make counts negative. *)
                -Counts[-Select[Negative][x]] 
                }, Total];
    
    (* Reconstruct the net multiset. *)
    Sort @ Flatten @ KeyValueMap[
        If[#2 == 0, 
            {}, (* Exact cancellation. *)
            (* Restore sign and multiplicity *)
            ConstantArray[Sign[#2] * #1, Abs[#2]] 
        ] &, tally]
    ]


(* Applies graded inhibition, drops negatives, squashes excitatory survivors. *)

ResolveNormal[x_List] := Module[{AE, AI, countsE, countsI},

    AE =  Select[Positive][x];
    AI = -Select[Negative][x]; 
    
    If[AI === {}, Return[Union[AE]]]; (* Fast path for pure boolean. *)
    
    countsE = Counts[AE];
    countsI = Counts[AI];
    
    Sort[Select[Keys[countsE], countsE[#] > Lookup[countsI, #, 0] &]]
    ]


(* Applies graded inhibition, drops negatives, keeps excitatory survivors. *)

ResolveGraded[x_List] := Module[{AE, AI, countsE, countsI},

    AE =  Select[Positive][x];
    AI = -Select[Negative][x];
    
    If[AI === {}, Return[Sort[AE]]]; (* Fast path for pure multisets. *)
    
    countsE = Counts[AE];
    countsI = Counts[AI];
    
    Sort[Flatten @ KeyValueMap[
        ConstantArray[#1, Max[0, #2 - Lookup[countsI, #1, 0]]] &, 
        countsE ]]
    ]


(* 
Transparently joins multisets. Preserves duplicates and negative signs. 
*)

MultisetBlockJoin[blocks : {__List}, dims : {__Integer}] := Module[{offsets},
    (* Calculate the starting index for each block *)
    offsets = FoldList[Plus, 0, Most[dims]];
    
    Flatten @ MapThread[
        Function[{sublist, offset},
            (* Shift absolute IDs to prevent collision, then restore the sign *)
            Sign[sublist] * (Abs[sublist] + offset)
        ],
        {blocks, offsets} ]
    ]


(* 
Transparently splits a joined multiset back into its original blocks. 
*)

MultisetBlockSplit[A_List, dims : {__Integer}] := Module[{limits, bounds},
    
    (* Calculate the ID boundaries for each block *)
    limits = FoldList[Plus, 0, dims];
    bounds = Partition[limits, 2, 1];
    
    Map[
        Function[{b},
            (* Select elements whose absolute ID falls in this block's range. *)
            Module[{chunk = Select[A, b[[1]] < Abs[#] <= b[[2]] &]},
                (* Remove the offset and restore the sign. *)
                Sign[chunk] * (Abs[chunk] - b[[1]])
            ]
        ], bounds ]
    ]


(* 
Join blocks {A1,A2,...} with dimensions {n1,n2,...} into a single list. 
Applies graded inhibition and drops inhibitory elements. 
Preserves multisets. 
*)
MultisetBlockJoinNormal[blocks : {__List}, dims : {__Integer}] := 
    Module[ {join, signs, AE, AI, countsE, countsI, netExcitatory},

    (* Add offsets to each block to ensure disjoint ID spaces. *)
    join = Function[{x}, Flatten[x + FoldList[Plus, 0, Most[dims]]]];
    
    signs = Sign /@ blocks;
    
    (* Extract and offset excitatory and inhibitory elements. *)
    AE = join[ MapThread[  Pick[#1, #2,  1]&, {blocks, signs}]];
    AI = join[ MapThread[ -Pick[#1, #2, -1]&, {blocks, signs}]];

    (* Count frequencies for graded subtraction. *)
    countsE = Counts[AE];
    countsI = Counts[AI];
    
    (* Graded Subtraction: Excitatory count minus Inhibitory count, 
        floored at zero. *)
    netExcitatory = Flatten @ KeyValueMap[
        ConstantArray[#1, Max[0, #2 - Lookup[countsI, #1, 0]]] &, 
        countsE
        ];
    
    (* Return the surviving excitatory multiset. Negatives are dropped. *)
    Sort[netExcitatory]
    ]


(*
Applies graded subtraction, drops negatives, and squashes the survivors. 
Outputs a flat, Boolean excitatory set.
*)

ResolveBlockJoinNormal[blocks: {__List}, dims : {__Integer}] := 
    Module[{join, signs, AE, AI, countsE, countsI, survivors},

    (* Add offsets to each block to ensure disjoint ID spaces *)
    join = Function[{x}, Flatten[x + FoldList[Plus, 0, Most[dims]]]];
    signs = Sign /@ blocks;
    
    (* Extract and offset excitatory and inhibitory elements. *)
    AE = join[MapThread[  Pick[#1, #2,  1]&, {blocks, signs}]];
    AI = join[MapThread[ -Pick[#1, #2, -1]&, {blocks, signs}]];

    (* Count frequencies for graded evaluation *)
    countsE = Counts[AE];
    countsI = Counts[AI];
    
    (* Squash: An element survives only if excitation  exceeds inhibition. *)
    survivors = Select[Keys[countsE], countsE[#] > Lookup[countsI, #, 0] &];
    
    Sort[survivors]
    ]


MemoryCapacity[{NA_, PA_}, {NB_, PB_}] := 
    If[NA == NB && PA == PB,
        (* Auto-associative. *)
        Round[ Log[2] * (NB * (NA-1) * (NA-2)) / (PB * (PA-1) * (PA-2) ) ],
        (* Hetero-associative. *)
        Round[ Log[2] * (NB * NA * (NA-1)) / (PB * PA * (PA-1) ) ]
    ]


MatchingThreshold[{NA_Integer, PA_Integer}, {NB_Integer, PB_Integer}] :=        
    Module[ {cap, T = 1},
        If[ PA <= 2 || PB <= 2, Return[0]]; (* Prevent division by zero. *)
        cap = MemoryCapacity[{NA, PA}, {NB, PB}];
        While[T < PA && cap^2 * Binomial[PA, T] Binomial[NA - PA, PA - T] / 
            Binomial[NA, PA] >= 1, T++]; 
        T
    ]                                                                           


(* General-purpose SHR encoder/decoder for symbolic expressions. *)

codec[config_Association] := Module[ {f, n, p, lex, T},

    {n, p} = config["hyperparameters"];
    
    (* Create global hash table shared by all instances with
    the same hyperparameters, enabling encode-decode roundtrips. *)
     
    lex = Circuit`codecLexicon[n, p];
    If[ ! MatchQ[ lex, _DataStructure], 
        lex = Circuit`codecLexicon[n, p] = CreateDataStructure["HashTable"]];
    lex["Insert", Null -> {} ];
    
    (* Use the auto-associative pattern matching threshold. *)
    T = MatchingThreshold[{n, p}, {n, p}]; 

    
    f[expr_] := Module[ {result, overlaps},
    
        (* Encoding *)
        If[ config["component"] === input,

            (* Bundle multiple tokens *)
            If[MatchQ[expr, _List], Return[Union[Flatten[f /@ expr]]]];
            
            If[MissingQ[lex["Lookup", expr]], 
                lex["Insert", expr -> Sort[RandomSample[Range[n], p]]]];
                
            Return[lex["Lookup", expr]];    
            ];
        
        (* Decoding *)
        If[expr === {}, Return[Null]];
        (* 
        This effectively implements an auto-associative memory for sets.
        A more scalable version would use an item memory for cleanup and
        a hash-based lookup for decoding.
        *)
        overlaps = Map[ Length[Intersection[#, expr]]&, Normal[lex]];

        (* Find all matches within the auto-associative matching threshold *)
        result = Keys[Select[overlaps, # >= T &]];
                
        Switch[ Length[result],
            0, Null,
            1, First[result],
            _, Sort[result]]
        
        ];
        
    <| "label" -> "SYM", "function" -> f, "clear" :> lex["KeyDropAll"] |>
    ]


(* 
Random category encoder.
Generates non-overlapping SHRs for K classes.
*)

category[config_Association] := Module[ {f, n, p, K, partitionsize, lex, T},

    {n, p} = config["hyperparameters"];
    
    (* Number of categories / classes. *)
    K = Lookup[config, "categories", Floor[n / p]];
    If[ K*p > n, Message[Circuit::category, K*p]];
    partitionsize = Floor[n / K];

    (* Create global hash table shared by all instances with
    the same hyperparameters, enabling encode-decode roundtrips. *)

    lex = Circuit`categoryLexicon[n, p, K];
    If[ ! MatchQ[ lex, _DataStructure], 
        lex = Circuit`categoryLexicon[n, p, K] = CreateDataStructure["HashTable"]];
    lex["Insert", Null -> {} ];
    
    (* Use the auto-associative pattern matching threshold (improve?) *)
    T = MatchingThreshold[{n, p}, {n, p}]; 

    f[expr_] := Module[ {result, overlaps},
    
        (* Encoding *)
        If[ config["component"] === input,

            If[!IntegerQ[expr] || Negative[expr] || expr >= K, Return[{}]]; 
        
            If[MissingQ[lex["Lookup", expr]], 
                lex["Insert", expr -> partitionsize*expr + Range[partitionsize]]];
            
            Return[Sort[RandomSample[lex["Lookup", expr], UpTo[p]]]];
            ];
        
        (* Decoding (same as codec) *)
        If[expr === {}, Return[Null]];
        overlaps = Map[ Length[Intersection[#, expr]]&, Normal[lex]];
        (* Find all matches within the auto-associative matching threshold *)
        result = Keys[Select[overlaps, # >= T &]];
        Switch[ Length[result],
            0, Null,
            1, First[result],
            _, Sort[result]]
        
        ];
        
    <| "label" -> "CAT", "function" -> f, "clear" :> lex["KeyDropAll"] |>
    ]


(* 
Clustering algorithm / decoder for sets. Assigns bin numbers 1,2,...
to every new cluster found.
*)
    
binning[config_Association] := Module[ {f, n, p, lex, T},

    {n, p} = config["hyperparameters"];
    
    (* Create global hash table shared by all instances with
    the same hyperparameters, enabling encode-decode roundtrips. *)

    lex = Circuit`binningLexicon[n, p];
    If[ ! MatchQ[ lex, _DataStructure], 
        lex = Circuit`binningLexicon[n, p] = CreateDataStructure["HashTable"]];
    lex["Insert", Null -> {}];
    
    (* Use the auto-associative pattern matching threshold. *)
    T = MatchingThreshold[{n, p}, {n, p}]; 

    (* Decoder only *)
    If[config["component"] === input,
        Message[Circuit::decoder, binning]; Return[<||>]];
        
    f[expr_] := Module[ {bins, overlaps},
        If[expr === {}, Return[0]];
        overlaps = Map[ Length[Intersection[#, expr]]&, Normal[lex]];
        bins = Keys[Select[overlaps, # >= T &]];
        
        Switch[ Length[bins],
            0, lex["Insert",  (bins = Length[overlaps]) -> expr]; bins,
            1, First[bins],
            _, Sort[bins]]
        ];
        
    <| "label" -> "BIN", "function" -> f, "clear" :> lex["KeyDropAll"] |>
    ]       


(* 
General-purpose SDR encoder for dense vectors. 
*)
    
vectorencoder[config_Association] := Module[ {f, n, p, R = <||>},

    {n, p} = config["hyperparameters"];

    (* Encoding only. *)
    If[ config["component"] === output,
        Message[Circuit::encoder, vectorencoder]; Return[ <||>];];

    f[list_List] := Module[ {d, sparsity, vec, min, max, span, 
                    scale, tolerance = 1.*^-10},

        d = Length[list];
        If[p <= 0 || d == 0, Return[{}]];
        
        sparsity = Lookup[config, "sparsity", N[1/Sqrt[d]]];
        
        (* Lazy initialization of random sparse binary matrix. *)
        If[ MissingQ[R[d]], 
            R[d] = SparseArray[(QuotientRemainder[
                RandomSample[Range[n*d], Round[n*d*sparsity]]-1, d] + 1) -> 1,
                {n, d}]
             ];

        vec = R[d] . list;
            
        (* Compute kWTA using rank p. *)
        (* Here we make no assumption about the bounds of the input data. *)
        {min, max} = MinMax[vec];
        span = max - min;
        scale = Max[Abs[min], Abs[max]];

        (* General-purpose dead switch. *)
        (* Reject if the array is zeros, or if the span is tiny relative 
            to the overall scale. *)
        If[scale == 0 || span <= tolerance * scale, Return[{}]];
        
        (* Standard kWTA *)
        Sort[Ordering[vec, -Min[p, Length[vec]]]]
        (* Note: This selects elements strictly in the original order,
        which may introduce a bias in certain datasets. *) 
        ];

    <| "label" -> "VEC", "function" -> f, "clear" :> (R = <||>) |>
    ]   


(* 
The textbook flyhash algorithm. 
*)
  
flyhash[config_Association] := 
    Module[{f, n, p, sparsity, R = <||>, restingpotential = <||>},
    
    {n, p} = config["hyperparameters"];

    sparsity = Lookup[config, "sparsity", 0.1];

    (* Encoding only. *)
    If[ config["component"] === output,
        Message[Circuit::encoder, flyhash]; Return[ <||>];];
    
    f[list_] := 
        Module[{d, c, centeredlist, vec, indices, min, max, span, 
                    scale, tolerance = 1.*^-10},
     
        d = Length[list];
        If[p <= 0 || d == 0, Return[{}]];
     
        (* Input centering. Removes baseline firing magnitude. *)
        centeredlist = list - Mean[list];
     
        (* Fixed fan-in projection matrix. *)
        If[MissingQ[R[d]],
      
            (* The number of glomeruli each Kenyon cell samples from. *)
            c = Max[1, Round[d * sparsity]]; 
      
            (* Generate c distinct random column indices for each row. *)
            indices = Flatten[Table[{i, j}, {i, n}, 
                {j, RandomSample[Range[d], c]}], 1];
      
            (* Fixed biological resting potential, as static tie breaker. *)
            R[d] = SparseArray[indices -> 1, {n, d}];
            restingpotential[d] = RandomReal[{0, tolerance * 0.01}, n];
            ];
     
        (* Pass signal, add the static resting potential to break ties. *)
        vec = R[d] . centeredlist + restingpotential[d];     
      
        (* General-purpose dead switch *)
        {min, max} = MinMax[vec];
        span = max - min;
        scale = Max[Abs[min], Abs[max]];
        If[scale == 0 || span <= tolerance * scale, Return[{}]];
     
        (* The APL neuron. Calculate standard kWTA. *)
        Sort[Ordering[vec, -Min[p, Length[vec]]]]
        ];

    <| "label" -> "FLY", "function" -> f, "clear" :> (R = <||>) |>
    ]  


(* 
At the beginning of each cycle, external input is pushed 
into the input's send slots. Input nodes have no "receive" slots.
A network may contain multiple input nodes.
Input data can be excitatory or inhibitory, or a mix.
The callback function serves as an optional encoder/preprocessing plugin.
*)
    
input[config_Association] := Module[ {plugin, pluginconfig},

    pluginconfig = 
        Append[ config, "hyperparameters" -> First[config["send_blocks"]]];

    plugin = 
        Lookup[config, "plugin", <| "function" -> Identity |>&] [pluginconfig];


    Join[ plugin, <|"size" -> 10,  "checks" -> {"input", "oneout"} |> ] 
    ]


(* 
At the end of each cycle, the output's input edges are read out.
Output nodes have no "send" slots.
A network may contain multiple output nodes.
Output data can be excitatory or inhibitory, or a mix.
The callback function serves as an optional decoder/postprocessing plugin.
*)

output[config_Association] := Module[ {plugin, pluginconfig},

    pluginconfig = 
        Append[ config, "hyperparameters" -> First[config["receive_blocks"]]];

    plugin = 
        Lookup[config, "plugin", <| "function" -> Identity |>&][pluginconfig];

    Join[ plugin, <|"size" -> 10, "checks" -> {"output", "oneinp"} |> ] 
    ]


(*
The delay node encapsulates multiple functionalities:

- signal delay
- block coding and re-coding
- proportional decimation (default: none)
- rate limiting  (default: none)
- temporal integration via internal capacity (default: none),
    using the plugin mechanism for update rules.

Multisets and inhibitory signals are handled transparently. 
*)

delay[config_Association] := 
    Module[ {f, plugin, pluginfactory, pop, dims1, dims2, absthreshold, decay, decimation, 
            absratelimit, abscapacity, latch, Xstate = {}},

    pluginfactory = Lookup[config, "plugin", replacement];
    plugin = pluginfactory[config];

    If[ pluginfactory === replacement,  plugin = KeyDrop[plugin, "label"]];

    dims1 = First /@ config["receive_blocks"];
    dims2 = First /@ config["send_blocks"];

    pop = Plus @@ Last /@ config["receive_blocks"];

    (* Drop inputs below threshold population. *)
    absthreshold = Round[pop * Lookup[config, "threshold", 0]];
    (* Limit size of query pattern. Default: unlimited. *)          
    absratelimit  = Round[pop * Lookup[config, "rate_limit", Infinity]];
    (* Pre-integration decay (leak rate). *)            
    decay = Lookup[config, "decay", 0];
    (* Temporal capacity, carry over from previous cycle. *)
    abscapacity = Round[ pop * Lookup[config, "capacity", 1]];              
    (* Proportional stochastic subsampling. *)
    decimation = Lookup[config, "decimate", 1];
    (* Latch option. *)
    latch = Lookup[config, "latch", False];


    f[blocks__List] := Module[ {X, Xdec},
        
        X = MultisetBlockJoin[ {blocks}, dims1];
        
        (* Enforce minimum population. *)
        If[Length[X] < absthreshold, X = {}];
        
        (* Rate limiting equally applies to positive and negative elements *)
        If[Length[X] > absratelimit, 
            X = Sort[RandomSample[X, absratelimit]]]; 

        (* Pre-integration stochastic decay. *)
        If[decay > 0, Xstate = 
            Sort[RandomSample[Xstate, Floor[(1 - decay) * Length[Xstate]]]]];

        (* Multiset subsampling applied to previous state. *)
        Xstate = Sort[RandomSample[Xstate, UpTo[abscapacity]]]; 
        
        (* Temporal integration via update rules. *)
        If[ ! (latch && Length[X] == 0), 
            Xstate = plugin["updaterule"][X, Xstate]]; 

        (* Proportional multiset subsampling applied to output *)
        Xdec = Xstate;
        If[decimation < 1,  
            Xdec = Sort[RandomSample[Xdec, Floor[decimation * Length[Xdec]]]]]; 

        Sequence @@ MultisetBlockSplit[Xdec, dims2]
        ];

    Join[ plugin, <| "function" -> f, "checks" -> {"arginp", "argout", "totaldim"},
        "fill" -> 13 |>]    
    ]


(* 
Plug-ins for update rules ("auto" component") 
and temporal integration ("delay" component).
*)


(* 
Replaces #2 with #1.
*)
replacement[_Association] :=
    <|"updaterule" -> (#1 &), "label" -> "\[FilledDownTriangle]", "size" -> 18 |>;

        
(* 
Multiset Complement[#2, #1]. Removes #1 from #2. Retains novel input.
*)
residual[_Association] :=
    <|"updaterule" -> (ResolveGraded[Multiset[#2, -#1]]&), "label" -> "\[FilledUpTriangle]", "size" -> 18 |>;

(* 
Multiset Complement[#1, #2]. Removes the #2 from #1. 
Isolates predicted elements.
*)
complement[_Association] :=
    <|"updaterule" -> (ResolveGraded[Multiset[#1, -#2]] &), "label" -> "\[EmptyDownTriangle]", "size" -> 20 |>;
    
(* 
Multiset SymmetricDifference[#1, #2].
Retains the symmetric difference between #2 and #1. 
*)
difference[_Association] :=
    <|"updaterule" -> (Abs[Multiset[#1, -#2]] &), "label" -> "\[EmptyUpTriangle]", "size" -> 20 |>;
    
(* 
Multiset Union[#1, #2]. Adds #1 to #2 via multiset aggregation.  
*)
augmentation[_Association] :=
    <|"updaterule" -> Multiset, "label" -> "\[Union]", "size" -> 14 |>;

(* 
Multiset Intersection[#1, #2].
Retains only the elements present in both #1 and #2.
*)
coincidence[_Association] :=
    <|"updaterule" -> (Module[{cY = Counts[#1], cX = Counts[#2]}, 
        Flatten @ KeyValueMap[ConstantArray[#1, Min[#2, Lookup[cX, #1, 0]]] &, cY]] &), 
        "label" -> "\[Intersection]", "size" -> 14 |>;

  
(*
Multiset Union of [ #1, permutation[#2]]
*)            
permutation[config_Association] := Module[{perm, dim},
    
    (* Total dimension needed to generate the permutation mapping. *)
    dim = First[Plus @@ config["receive_blocks"]];
    
    (* Initialize fixed, instance-specific permutation array. *)
    perm = RandomSample[Range[dim]];

    <| "updaterule" -> (Multiset[#1, Sort[Sign[#2] * perm[[Abs[#2]]]]] &), 
        "label" -> "\[Pi]", "size" -> 16 |>
    ]        


(* 
Auto-associative memory component. 

Note: There is a wide range of possible learning, subsampling, retrieval 
and update rules for auto-associative memory. This prototype captures the 
geneneric cases. Modify as needed.
*)

auto[config_Association] := 
    Module[ {f, plugin, dims, pop, params, M, min, max, 
         absratelimit, decimation},
        
    (* Update rule *)           
    plugin = Lookup[config, "plugin", replacement][config];

    dims = First /@ config["receive_blocks"];
    params = Plus @@ config["receive_blocks"];
    pop  = Last[params];

    (* Limit size of query pattern. Default: unlimited. *)          
    absratelimit  = Round[Lookup[config, "rate_limit", Infinity] * pop];
    (* Proportional stochastic subsampling for memory query. *)
    decimation = Lookup[config, "decimate", 1];
    
    (* Learning thresholds. These apply to the total state, aggregated
    from all blocks. Modify this logic as needed. The option values are
    given as scaling factors. *)

    min = max = pop; (* defaults *)
    If[ NumberQ[config["learn"]], min = max = Round[config["learn"]*pop] ];
    If[ MatchQ[config["learn"], {__?NumberQ,__?NumberQ}], 
        {min, max} = Round[config["learn"] * pop]];
    
    (* Memory with identical input and output parameters. *)
    M = Memory[Join[config, 
        <| "A_parameters" -> params, "B_parameters" -> params |> ]];    
        
    f[blocks__List] := Module[ {A, X, Y, Xdec},

        (* Join partitions and apply inhibition. *)
        A = X = ResolveBlockJoinNormal[{blocks}, dims];
        
        (* Limit the rate of incoming positives. *)
        If[Length[X] > absratelimit, 
            X = Sort[RandomSample[X, absratelimit]]];

        (* Note: Proportional decimation may result in an empty list. *)
        Xdec = If[decimation < 1, 
            Sort[RandomSample[X, Floor[decimation * Length[X]]]], X];
                
        (* Always learn if "learn"->True. Typically used for hyperassociations. *)
        If[config["learn"], M["store"][A]];

        (* Memory retrieval with decimated and rate-limited input state. *)
        Y = M["retrieve"][Xdec];

        (* Learn input A if retrieval fails *)
        If[Y === {} && Length[A] >= min && Length[A] <= max,
            If[config["learn"] =!= True, M["store"][A]]; Y = A];

        X = Union[plugin["updaterule"][Y, X]]; 

        Sequence @@ MultisetBlockSplit[X, dims] 
        ]; 
        
        
    Join[ plugin, <| "function" -> f, "checks" -> {"arginp", "argout", "ident"}, 
        "shape" -> "square", "fill" -> 8 |>, M]
    ]


(*
Temporal-associative memory component. 
Learns higher-order sequences on the fly and predicts the next token.
A hybrid between auto-associative and hetero-associative architectures.
Uses the same temporal integration plugins and parameterization as "delay".
*)

temporal[config_Association] := 
    Module[ {f, M, plugin, dims, params, pop, absthreshold, decay, decimation, 
            absratelimit, abscapacity, latch, Xstate = {}, Xdec, prediction = {}},

    plugin = Lookup[config, "plugin", replacement][config];

    dims = First /@ config["receive_blocks"];
    params = Plus @@ config["receive_blocks"];
    pop  = Last[params];
    
    (* Drop inputs below threshold population. *)
    absthreshold = Round[pop * Lookup[config, "threshold", 0]];
    (* Limit size of query pattern. Default: unlimited. *)          
    absratelimit  = Round[pop * Lookup[config, "rate_limit", Infinity]];
    (* Pre-integration decay (leak rate). *)            
    decay = Lookup[config, "decay", 0];
    (* Temporal capacity, carry over from previous cycle. *)
    abscapacity = Round[ pop * Lookup[config, "capacity", 1]];              
    (* Proportional stochastic subsampling for memory query. *)
    decimation = Lookup[config, "decimate", 1];
    (* Latch option. *)
    latch = Lookup[config, "latch", False];

    (* Memory with identical input and output parameters. *)
    M = Memory[Join[config, 
        <| "A_parameters" -> params, "B_parameters" -> params |> ]];    


    f[blocks__List] := Module[ {X},
        
        X = MultisetBlockJoin[ {blocks}, dims];
        
        (* Enforce minimum population. *)
        If[Length[X] < absthreshold, X = {}];
        
        (* Rate limiting equally applies to positive and negative elements *)
        If[Length[X] > absratelimit, 
            X = Sort[RandomSample[X, absratelimit]]]; 

        (* Pre-integration stochastic decay. *)
        If[decay > 0, Xstate = 
            Sort[RandomSample[Xstate, Floor[(1 - decay) * Length[Xstate]]]]];

        (* Learn state -> current input if prediction was incorrect. *)
        If[ prediction =!= X, 
            M["store"][ResolveNormal[Xstate], ResolveNormal[X]]];

        (* Multiset subsampling applied to previous state. *)
        Xstate = RandomSample[Xstate, UpTo[abscapacity]]; 
        
        (* Temporal integration via update rules. *)
        If[ ! (latch && Length[X] == 0), 
            Xstate = plugin["updaterule"][X, Xstate]]; 

        (* Proportional multiset subsampling applied to the integrated state,
         to break symmetry during retrieval. *)
        Xdec = Xstate;
        If[decimation < 1,  
            Xdec = Sort[RandomSample[Xdec, Floor[decimation * Length[Xdec]]]]]; 

        (* Predict next token. *)
        prediction = M["retrieve"][ResolveNormal[Xdec]];

        Sequence @@ MultisetBlockSplit[prediction, dims]
        ];

    Join[ plugin, <| "function" -> f, "checks" -> {"arginp", "argout", "ident"},
        "shape" -> "square", "fill" -> 14 |>]   
    ]


(* 
Vanilla hetero-associative node for supervised learning A -> B.
B is the first input argument.
A is given by the rest of the input arguments. Can be block-coded.
Always learns if B =!= {}.
*)

associator[config_Association] := 
    Module[ {f, Adims, M},

    Adims = First /@ Rest[config["receive_blocks"]]; (* Dimensions of input blocks A *)
    
    (* Memory instance: *)
    M = Memory[Join[config, <|
        "A_parameters" -> Plus@@Rest[config["receive_blocks"]], 
        "B_parameters" -> First[config["receive_blocks"]] |> ]];
    
    f[B_List, blocks__List] := Module[{X, Y},
        
        (* Apply and drop inhibitory elements. *)
        X = ResolveBlockJoinNormal[{blocks}, Adims];
        Y = ResolveNormal[B];
        
        If[Y === {}, Return[ M["retrieve"][X]]]; (* Test. *)
        
        M["store"][X, Y]; 
        {}  
        ];   
        
    Join[ <| "function" -> f, "checks" -> { "dimfirst", "oneout"},
        "shape" -> "square", "fill" -> 11, "size" -> 18, "label"-> "\[FilledRightTriangle]\[FilledCircle]" |>, M]
    ]


(* 
Heteroencoder A -> B. A may be partitioned. Has no input for B.
*)

heteroencoder[config_Association] := 
    Module[ {f, dims, M},
    
    dims = First /@ config["receive_blocks"];
    
    M = Memory[ Join[config, <|
        "A_parameters" -> Plus @@ config["receive_blocks"],
        "B_parameters" -> First[config["send_blocks"]] |>  ]];
    
    f[blocks__List] := Module[{X, Y},
        X = ResolveBlockJoinNormal[{blocks}, dims];
        
        Y = M["retrieve"][X];
        
        (* Generate new SHR encoding if retrieval fails. *)
        If[Y === {}, 
            Y = Sort[RandomSample[Range[#1], #2]]& @@ First[config["send_blocks"]]];
        
        (* Always learn, broadening X as an attractor. *)
        M["store"][X, Y];
            
        Y
        ]; 
            
    Join[ <| "function" -> f, "checks" -> {"arginp", "oneout"}, 
        "shape" -> "square",  "fill" -> 11,  "size" -> 18,
        "label" -> "\[FilledLeftTriangle]\[FilledRightTriangle]" |>, M]
    ]


(*
Generates a prediction based on the current input (slot #1) 
and context (slots #2,...).
Automatically learns the correct prediction in the following cycle.
To do so, the node remembers its context from the previous cycle.
*)

predictor[config_Association] := 
        Module[ {f, M, X = {}, prediction = {}, decimation,
                        itemconfig, contextconfig},

    decimation = Lookup[config, "decimate", 1];
    itemconfig = First[config["receive_blocks"]];
    contextconfig = Rest[config["receive_blocks"]]; (* Context may be partitioned *)
    
    M = Memory[Join[config, <|
        "A_parameters" -> Plus @@ contextconfig,
        "B_parameters" -> itemconfig   |> ]]; 
    
    f[item_List, blocks__List] := Module[{Xdec},
    
        (* State X from previous cycle *)
        (* Always learn. Also when the prediction was correct. *)
        If[ prediction =!= X, 
            M["store"][X, ResolveNormal[item]]];
        
        X = ResolveBlockJoinNormal[ {blocks}, First /@ contextconfig];

        Xdec = Sort[RandomSample[X, Floor[decimation * Length[X]]]];
        
        prediction = M["retrieve"][Xdec]
        ]; 


    Join[ <| "function" -> f, "checks" -> {"dimfirst", "oneout"}, 
        "shape" -> "square", "fill" -> 11, "size" -> 18,
        "label" -> "\[FilledRightTriangle]\[FilledRightTriangle]" |>, M]
    ]


(* 
Random noise generator
*)
    
noise[config_Association] := Module[ {f},

    f[blocks___] := 
        Sort[RandomSample[Range[#1], #2]]& @@ First[config["send_blocks"]]; 
    
    <| "function" -> f, "checks" -> {"input", "oneout"}, "label" -> "~",
        "fill" -> 13, "size" -> 18 |>
    ]   


(* 
Plug-in for importing an embedded circuit from a json file.
*)
                
file[config_Association] := 
    Module[ {file, circ, sub, rescale},
    
    file = config["name"];
    If[ MissingQ[file], Message[Circuit::nofile, config]; Return[<||>]];
    
    (* File name extension may be omitted. *)
    If[ !StringMatchQ[file, "*.json"], file = file <> ".json"];
    
    If[ FindFile[file] === $Failed,
        Message[Circuit::findfile, file]; Return[<||>] ];
        
    circ = Import[file, "Circuit"];
    
    (* Rescale hyperparameters of embedded circuit. *)
    If[ MatchQ[config["scale"], {_Integer, _Integer}] &&
        MatchQ[circ["hyperparameters", "default"], {_Integer, _Integer}],
            
        (* Scaling factors, as a pair of integers or rationals. *)
        rescale = config["scale"] / circ["hyperparameters", "default"]; 

        (* Rescale: *)
        circ["hyperparameters"] //= Map[Round[# * rescale] &];
        ];
            
    sub = Circuit[circ];
    
    sub["label"] = StringTake[FileBaseName[file],UpTo[5]];
    
    sub (* Includes "clear". *)
    ]


(* 
Plug-in for embedded a precompiled circuit
*)

shared[config_Association] := 
    Module[ {name, sub},
    
    (* Circuit instance is registered under the "name" property. *)
    name = config["name"];
    If[ MissingQ[name], Message[Circuit::name, config]; Return[<||>]];
    
    sub = Lookup[ Circuit`Registry, name];
    If[ MissingQ[sub], Message[Circuit::circuit, name]; Return[<||>]];

    sub["label"] = name;

    KeyDrop[sub, "clear"] (* Do not propagate "clear". *)
    ]


(* 
Embedded circuit component.
Multiple circuit components can reference the same embedded instance.
*)
                
circuit[config_Association] := 
    Module[ {f, plugin, params},
    
    plugin = Lookup[config, "plugin", shared] [config];
    
    If[ plugin === <||>, Return[<||>]];
    
    params = {plugin["receive_blocks"], plugin["send_blocks"]};
    
    If[ {config["receive_blocks"], config["send_blocks"]} =!= params,
            Message[Circuit::embedded, params]; plugin = <||>];
        
    (* Raw evaluation, bypassing encoding and decoding. *)
    f[blocks__] := plugin["function"][blocks];
    
    Join[plugin, <| "function" -> f, "checks" -> {"arginp", "argout"}, 
                            "shape" -> "square", "size" -> 9|>]
    ]                   


(* Dataflow tags and their JSON names. *)
Circuit`TagMap = <|

"multiset"        -> "+", (* only multisets may contain negatives *)
"inhibit"         -> "-", (* if - then automatically + *) 
"permute"         -> "\[Pi]", (* permuation, unique per pathway *)
"noise"           -> "~", (* random noise if empty, else clear elements *)
"rate_limit"      -> "R", (* stochastic rate limiting up to hyperparameter P *)
"threshold"       -> "T", (* clearing data below hyperparameter population P *)
"veto"            -> "X", (* clears all paths if any X path is non-empty *)
"mandatory"       -> "*", (* clears all paths if any * path is empty *)
"priority"        -> "!", (* clears all non-! paths if any ! path is non-empty *) 
"dependency"      -> "&", (* clears & paths if any non-& path is empty (AND) *)
"fallback"        -> "|", (* clears | paths if any non-| path is non-empty (NOR) *)
"barrier"         -> "=", (* clears = paths if any = path is empty *)
"kwta_excitatory" -> "K", (* top-k of positives *)
"kwta_absolute"   -> "k", (* top-k of all elements *)
"log"             -> "?", (* print log of path contents *)
"show_slot"       -> "#", (* render slot number *)
"show_dimension"  -> "$", (* render N *)
"show_population" -> "%"  (* render P *)

|>;

(* Note: temporal gating @... is handled separately. *)


Circuit`Registry = <||>; (* To keep track of embedded (nested) circuits. *)


Circuit[expr_] := Module[ 
        { 
        assoc, dispatch, 
        
        (* Exposed in dispatch association. *)
        f, multiplex, schematics, clear, receiveparams = {}, sendparams = {}, 
    
        (* Compiler. *)
        compile, 
                
        (* Static compiled data. *) 
        nodes = <||>,
        pathways = <||>,
        
        preprocess  = {},        (* Encoding. One entry per input node. *)
        postprocess = {},        (* Decoding. One entry per output node. *) 
        
        inputslots = {},
        outputedges = {},
        
        permutations = <||>,

        (* Runtime functions. *)
        evaluate, patheval, pathmerge, componenteval,
                                    
        (* This circuit's clock. *)
        tick = 0,

        (* Slot state buffers. *)
        nextstate = <||>,
        currentstate = <||>,    
    
        nodeid = 0, edgeid = 0 
        },
        
    assoc = Switch[ expr,
    
        (* Raw circuit spec, e.g. imported from JSON or DFD. *)
        _Association /; KeyExistsQ[expr, "dataflow"], expr,
        
        (* Convert from DFD (supported only in Mathematica). *)
        _List, FromDFD[expr],
                
        (* JSON string. *)
        _String /; StringContainsQ[expr, "dataflow"], FromJSON[expr],

        _, Message[Circuit::circargs, expr]; Return[<||>];
        ];  
        
                
    (* Compile a component. This triggers compilation of incoming pathways. *)
    compile[node_Association] := Module[
        { config, compilepathway, slotparams, fillparams}, 
                
        ++ nodeid;
        
        compilepathway[receive_] := Module[{e, tag},
            
            (* List of pathways denotes merging.  *)
            If[ ListQ[receive], Return[ compilepathway /@ receive]];

            ++edgeid;

            tag = MemberQ[e["tags"], #] &;

            e = <|
                "to_node_id" -> nodeid, (* "from_node_id" added after all nodes are compiled *)
                "label" -> "", (* default. *)
                "tags" -> {} (* default. *)
                |>;
                
            Switch[ receive,
                _Integer, e["slot"] = receive,
                _Association, e = Join[e, receive],
                _, Message[Circuit::invedge,e]; <||>
                ];
                
            e["hyperparameters"] = slotparams[e["slot"]];

            (* Helpful compiler: If "inhibit" then always "multiset" *)
            If[tag["inhibit"], 
                e["tags"] = Union[e["tags"], {"multiset"}]];                
            
            (* Register this pathway and initialize buffer. *)
            pathways[edgeid] = e;
            currentstate[e["slot"]] = {};
                 
            (* Initialize permutation for any pathway, even if it's not used. *)
            permutations[edgeid] = 
                RandomSample[Range[e["hyperparameters"][[1]]]];
                     
            edgeid   
            ];      
        
        (* Per-slot hyperparameter settings. *)
        slotparams[slot_Integer] := Module[{hyperparameters, default, np},
            hyperparameters = Lookup[assoc, "hyperparameters", <||>];
        
            default = Lookup[hyperparameters, "default", None];
            np = Lookup[hyperparameters, slot, default];
        
            If[ !MatchQ[np, {_Integer, _Integer}], 
                    Message[Circuit::noparam, slot]; {100, 3}, np]
            (* Default to {100, 3} to silence downstream error messages. *)
            ];

        fillparams[x_] := Switch[x, 
            _List, 
                Module[ {list, dims},
                    list = fillparams /@ x; (* Recursion *)
                    dims = list[[All,1]];
                    If[! SameQ @@ dims, Message[Circuit::dimbundle, x]];
                    (* Combine pathway dimensions. *)
                    {dims[[1]], Min[list[[All, 2]]]}
                    ],
            _Integer, 
                slotparams[x],
            _Association, 
                slotparams[x["slot"]],
            _, Message[Circuit::routing, x]; Nothing 
            ];
                                                                            
        (* Base configuration for each node. Includes per-node user settings. *)
        config = Join[ node, 
            <|  
            "receive_paths" -> compilepathway /@ Lookup[node, "receive", {}],
        
            "node_id" -> nodeid, (* Used only for schematics visualization. *)
            
            (* Extract slot numbers *)
            "send_slots" -> Cases[Lookup[node, "send", {}], _Integer],
            
            (* Reference parameters, used for rescaling embedded circuits. *)
            "scale" -> assoc["hyperparameters", "default"],

            "receive_blocks" -> fillparams /@ Lookup[node, "receive", {}],
            "send_blocks" -> fillparams /@ Lookup[node, "send", {}]
            |>];
            
        (* Instantiate the component, re-append config to ensure 
            user settings survive *)
        If[ ! MatchQ[Lookup[node, "component"], _Symbol], 
            Message[Circuit::component, Lookup[node, "component"]]; Return[]];  
            
        config = Join[Lookup[node, "component"][config], config];       

        (* Prepend default visualization settings. *)
        config = Join[<| "shape" -> "circle", "label" -> "", "fill" -> 6, 
            "color" -> 0, "size" -> 12 |>, config];

        (* Prepend default functional settings. *)
        config = Join[<| "checks" -> {}, "function" -> Null |>, config];

        (* Static error checking, based on component-supplied "checks". *)
        If[ ! Circuit`check[#][config["receive_blocks"], config["send_blocks"]], 
            Message[MessageName[Circuit, #], config]] & /@ config["checks"];

        (* Register input flow. *)
        If[Lookup[node, "component"] === input,  
            (* Register encoder. *)
            AppendTo[preprocess, config["function"]];
            AppendTo[inputslots, First[config["send_slots"]]];
            AppendTo[receiveparams, First[config["send_blocks"]]];

            config["label"] //= StringReplace["#" -> ToString[Length[inputslots]]]
            ];

        (* Register output flow. *)
        If[Lookup[node, "component"] === output, 
            (* Register decoder. *)
            AppendTo[postprocess, config["function"]];
            AppendTo[outputedges, First[config["receive_paths"]]];
            AppendTo[sendparams, First[config["receive_blocks"]]];

            config["label"] //= 
                StringReplace["#" -> ToString[Length[outputedges]]]
            ];
                
        (* Register this component. *)
        nodes[nodeid] = config;
        ];
    
    
    (* Compile the nodes. The nodes compile their inbound pathways. *)
    Scan[compile, Lookup[assoc, "dataflow", {}]]; 

    (* Link the edges. Needed for graph rendering. *)
    Module[ {slots, dupes, links = <||>, getlink},

        (* Duplicate sendslots? *)
        slots = Flatten[#["send_slots"] & /@ Values[nodes]];
        dupes = Keys[Select[Counts[{slots}], # > 1 &]];
        If[ Positive[Length[dupes]], Message[Circuit::dupslots, dupes]];

        (* Gather link table. *)
        Scan[ Function[c, (links[#] = c["node_id"]) & /@ c["send_slots"]], 
            nodes];

        (* Add "from_node_id" value to each pathway. *)
        pathways = Prepend[#, "from_node_id" -> Lookup[links, #["slot"], 
            Message[Circuit::routing, #["slot"]]; 0]] & /@ pathways;        
        ];

    (* 
    Component evaluation. Triggers evaluation of inbound pathways.
    *)
    componenteval[config_Association] := Module[ {x, result},

        (* Input and output nodes are not evaluated.*)      
        If[ MatchQ[config["component"], input | output], Return[]];
        
        (* Evaluate received pathways and latch inputs. *)
        x = pathmerge /@ config["receive_paths"];
            
        (* This component's callback function. *)
        result = config["function"][Sequence @@ x]; 

        (* Copy results to output slot. *)
        MapThread[(nextstate[#1] = #2) &, {config["send_slots"], {result}}]
        ];
                                                
    (* 
    Pathway evaluation.
    During each cycle, this is triggered either by component evaluation
    (except output components), or explicitly when collecting outputs.*)

    patheval[id_Integer] := 
        Module[ {e, x, tag, gating, n, p},
        
        tag = MemberQ[e["tags"], #] &;
        
        e = pathways[id];
        
        {n, p} = e["hyperparameters"];
        
        x = currentstate[e["slot"]];
        
        (* Runtime check. *)
        If[! VectorQ[x, IntegerQ] || MemberQ[x, 0], 
            Message[Circuit::integer, x]; Return[{}]];
                
        (* Process temporal gating (tags of the form @10101). *)
        gating = Lookup[e, "gating", {1}];
        If[gating[[Mod[tick - 1, Length[gating]] + 1]] == 0, Return[{}]];
        
        (* 
        Inhibition transformer. Flips positives to negatives.
        This is the ONLY source of inhibition. 
        Normal pathways apply inhibitory elements and then drop them. 
        This enforces Dale's law. 
        *)
        If[tag["inhibit"], x = -Abs[x]];
        
        (*
        Carrier: forward resolved multiset.
        Boundary: untagged edges force resolution to a flat, excitatory set. 
        *)
        x = If[tag["multiset"], Multiset[x], ResolveNormal[x]];

        (* Permutation. *)
        If[tag["permute"], x = Sort[Sign[x] * permutations[id][[Abs[x]]]] ];
    
        (* Rate limiting. *)
        If[tag["rate_limit"], x = Sort[RandomSample[x, UpTo[p]]]];

        (* Thresholding. *)
        If[tag["threshold"] && Length[x] < p, x = {}];

        (* Random noise if x = {}, else clear x *)
        If[tag["noise"], x = If[x === {}, Sort[RandomSample[Range[n], p]], {}]];

        (* Print value to console. *)
        If[tag["log"], Print["Slot ", e["slot"], ": ", x]];
            
        x
        ];
        

    (* 
    Merge pathways. Also apply this if there's only one pathway.
    *)
    pathmerge[id_Integer] := pathmerge[ {id}];
    
    pathmerge[list : {__Integer}] := 
        Module[ {x, paths, tags, merged},

        x = patheval /@ list;
        paths = pathways /@ list;
        
        (* List of Booleans with the same length as x. *)
        tags = Function[t, MemberQ[Lookup[#, "tags", {}], t] & /@ paths];       
        (* Logical gating. *)

        If[MemberQ[Pick[x, tags["veto"]], {__}], Return[{}]];

        If[MemberQ[Pick[x, tags["mandatory"]], {}], Return[{}]];

        If[MemberQ[Pick[x, tags["priority"]], {__}], 
            x = MapThread[If[#1, #2, {}] &, {tags["priority"], x}] ];

        If[MemberQ[Pick[x, Not /@ tags["dependency"]], {}], 
            x = MapThread[If[#1, {}, #2] &, {tags["dependency"], x}] ];
            
        If[MemberQ[Pick[x, Not /@ tags["fallback"]], {__}], 
            x = MapThread[If[#1, {}, #2] &, {tags["fallback"], x}] ];

        If[Or @@ tags["barrier"] && MemberQ[Pick[x, tags["barrier"]], {}],
             x = MapThread[If[#1, {}, #2]&, {tags["barrier"], x}]];

        (* Aggregate the survivors. *)
        merged = Multiset[x];
    
        (* Optional kWTA. *) 
        If[Or @@ tags["kwta_excitatory"] || Or @@ tags["kwta_absolute"],
            Module[{k, U, tally, rankedmax},
            
                k = Min[#["hyperparameters"][[2]] & /@ paths ];
            
                (* "K" forces excitatory only. 
                    "k" evaluates absolute saliency. *)
                U = If[Or @@ tags["kwta_excitatory"], 
                    Select[Positive][merged], merged];
            
                tally = SortBy[Tally[U], Last];
            
                (* Force consensus: k elements must appear at least twice. *)
                merged = {};
                If[Length[tally] >= k, rankedmax = tally[[-k, 2]];
                    If[ rankedmax >= 2, merged = Sort[Select[tally, 
                        Last[#] >= rankedmax &][[All, 1]]]]]
            ]
            ];  
                
        merged
        ];
        
    (* Raw tick-tock evaluation cycle. *)
    evaluate[x_List] := Module[ {},
        (* Advance this circuit's time. Used for multiplexing. *)
        tick ++;
        
        (* Optional clean-slate reset of nextstate, retaining all keys. 
        Safeguard for a scenario where a component fails to update nextstate. 
        nextstate = Map[{} &, currentstate];
        *)

        (*  Push external inputs to both slot buffers:
              currentstate: for downstream pathway evaluation,
              nextstate: because input components do not evaluate.
         *)
        MapThread[(currentstate[#1] = nextstate[#1] = #2) &, {inputslots, x}];

        (* Evaluate nodes, reading from currentstate, updating nextstate. *)
        componenteval /@ nodes;

        (*  Snapshot state, preparing for output extraction and next cycle. *)
        currentstate = nextstate;
                
        (* Gather outputs, reading from currentstate. *)
        pathmerge /@ outputedges
        ];

    (* 
    Raw evaluation with optional multiplexing.
    Operates on preprocessed (encoded) data.
    Used directly by embedded circuits.
    *)  
    multiplex[x_List] := Module[ {y = {}, mult},
    
        If[ Length[x] =!= Length[inputslots], 
            Message[Circuit::sequence, inputslots]; Return[Sequence[]]];
                
        (* Multiplexed evaluation of encoded data *)
        mult = Lookup[Lookup[assoc, "options", <||>], "multiplex", {1}];
        
        If[ ! MatchQ[ mult, {(0 | 1) ..}],
            Message[Circuit::multiplex, mult]; mult = {1}];
        
        Do[y = If[gate === 1, evaluate[x], 
            evaluate[{} & /@ x]], {gate, mult}];
            
        y (* Can be empty if all gates are zero. *)
        ];          


    (* 
    Top-level function corresponding to the circuit.
    Each call, even with empty inputs, triggers one execution cycle. 
    The function has a variable number of arguments which must match
    the number of input nodes.
    It returns a Sequence of expressions, each corresponding to an
    output node. The ordering of input and output arguments corresponds 
    to the order in which input and output nodes appear in the dataflow
    description.    
    *)  
    f[blocks___] := Module[ {x, y},
    
        (* Wrap input sequence into a list. *)
        x = {blocks};
    
        (* Number of external input slots must match number of input nodes. *)
        If[ Length[x] =!= Length[inputslots], 
            Message[Circuit::sequence, inputslots]; Return[{}]];
                
        (* Preprocess (encode). *)
        x = MapThread[Construct, {preprocess, x}];
            
        (* Main circuit function. *)
        y = multiplex[x];       
        
        (* Postprocess (decode). *)
        y = MapThread[Construct, {postprocess, y}]; 
            
        (* Strip list, return as sequence. *)
        Sequence @@ y
        ];
        
    (* Visualization. *)
    schematics[opts___Rule] := Module[ 
        { 
        graph, 
        layout = {"LayeredDigraphEmbedding", "Orientation" -> Left},
        coords = Automatic, radius = 0.4, colorscheme, 
        font = "Inter Variable", edgetag, display
        },
            
        (* Color palette lookup. *)
        colorscheme[c_] := Lookup[Circuit`color, c, c];
        
        (* Convert tag names to single-character display labels. *)
        display = Reverse[Circuit`TagMap];
                            
        edgetag[a_Association] := Module[ {str, rules, gatingstr = ""},

            If[KeyExistsQ[a, "gating"],
                gatingstr = "@" <> StringJoin[ToString /@ a["gating"]]];

            str = Lookup[a, "label", ""] <> 
                StringJoin[Lookup[display, #, ""]& /@ Lookup[a, "tags", {}]] <>
                gatingstr;
            
            rules =                 {
                display["show_slot"] -> ToString[a["slot"]],
                display["show_dimension"] -> ToString[a["hyperparameters"][[1]]],
                display["show_population"] -> ToString[a["hyperparameters"][[2]]]
                };
            
            StringReplace[str, rules]
            ];
        
        (* Compile graph *)
        graph = DirectedEdge[#["from_node_id"], #["to_node_id"], 
                edgetag[#]]& /@ Values[pathways];
        
        (* Force path graphs to render on a single line. *)
        If[PathGraphQ[SimpleGraph[graph]],
            coords = (# -> {#, 0})& /@ Range[Length[nodes]]];
            
        Graph[ graph,
            PlotTheme -> "Classic",
            BaseStyle -> {FontFamily -> font, Opacity[1]},
            
            VertexSize -> radius,  
            
            VertexShapeFunction -> Normal[Replace[#["shape"], 
                {"none" -> None, s_String :> Capitalize[s]}] & /@ nodes],
                                    
            VertexStyle -> Normal[
                Directive[colorscheme[#["fill"]], 
                EdgeForm[Directive[colorscheme[1], AbsoluteThickness[0.15]]],
                Opacity[1]]& /@ nodes],

            VertexLabels -> Normal[ Placed[Style[#["label"], 
                colorscheme[#["color"]],

FromDFD[dataflow_List] := Module[
    {rules, hyperparams = <||>, rootopts = <||>, customopts = <||>, 
     nodes, components, parsenode, parsepath, parsemerge},
    
    (* Include schema reference at the root *)
    AppendTo[rootopts, 
        "$schema" -> "https://creatingintelligence.org/schemas/circuit-v1.json"];

    (* Extract all rules from the dataflow. *)
    rules = Select[dataflow, Head[#] === Rule &];
    
    (* Sort integers/blanks to hyperparameters, specific keys to root, and others to options. *)
    Scan[ Function[{rule}, Which[
        rule[[1]] === _, 
            AppendTo[hyperparams, "default" -> rule[[2]]],
    
        IntegerQ[rule[[1]]], 
            AppendTo[hyperparams, rule[[1]] -> rule[[2]]],
        
        MemberQ[{"version", "metadata", "$schema"}, rule[[1]]],
            AppendTo[rootopts, rule[[1]] -> rule[[2]]],
        
        rule[[1]] === "multiplex" && StringQ[rule[[2]]],
            AppendTo[customopts, "multiplex" -> (If[# === "1", 1, 0] & /@ 
                    Characters[rule[[2]]])],
        
        StringQ[rule[[1]]], 
            AppendTo[customopts, rule[[1]] -> rule[[2]]],

        True, 
            AppendTo[customopts, ToString[rule[[1]]] -> rule[[2]]]
        ]], rules
    ];
    
    (* Extract circuit components. *)
    nodes = Select[dataflow, Head[#] =!= Rule &];
    
    (* Parse individual path strings into structured JSON objects. *)
    parsepath[p_] := Module[
        {s = ToString[p], match, slot, tags, tagmap, 
            found, result = <||>, tagpattern},
        
        match = StringCases[s, StartOfString ~~ n:DigitCharacter.. ~~ Whitespace... ~~ t:___ ~~ EndOfString :> {ToExpression[n], t}];
        
        If[match === {}, 
            tagpattern = (Alternatives @@ Join[Values[Circuit`TagMap], {" "}] | ("@" ~~ DigitCharacter..))..;
            If[StringMatchQ[s, tagpattern], match = {{1, s}}, Return[s]]
            ];
        
        {slot, tags} = match[[1]];
        If[tags === "", Return[slot]];
        
        result["slot"] = slot;
        
        If[StringContainsQ[tags, " "],
            {result["label"], tags} = StringSplit[tags, " ", 2]];
        
        With[{g = StringCases[tags, "@" ~~ val:(DigitCharacter..) :> val]},
            If[Positive[Length[g]],
                result["gating"] = If[# === "1", 1, 0] & /@ Characters[g[[1]]];
                tags = StringReplace[tags, "@" ~~ g[[1]] -> ""]; 
                ]
            ];
        
        tagmap = Association[Reverse /@ Normal[Circuit`TagMap]];
        tagmap["P"] = "permute"; (* Legacy support. *)

        found = Values[KeyTake[tagmap, Characters[tags]]];
        If[Positive[Length[found]], result["tags"] = found];
        
        result
        ];

    parsemerge[l_List] := parsepath /@ l;
    parsemerge[p_] := parsepath[p];

    parsenode[compExpr_[send___][receive___]] := Module
        [
        {comp, argslist, plugin = Null, opts = <||>},
        
        comp = Head[compExpr]; 
        argslist = List @@ compExpr; 
        
        Do[ Which
            [Head[arg] === Rule, AppendTo[opts, arg],
            True, plugin = arg ], 
                {arg, argslist}
            ];
        
        DeleteCases[ 
            Join[<|
            "component" -> comp, 
            "plugin" -> plugin,
            "send" -> Map[parsemerge, Flatten[{send}, 1]],
            "receive" -> Map[parsemerge, {receive}]
                |>, opts], 
            Null | {} | <||> 
            ]
    ];
        
    components = parsenode /@ nodes;
    If[Positive[Length[customopts]], AppendTo[rootopts, "options" -> customopts]];

    Join[rootopts, <|"hyperparameters" -> hyperparams, "dataflow" -> components|>]
    ]


ToDFD[expr_Association] := Module[
    {hyperparams, hprules, topoptsrules, dataflow, buildpath, 
     buildmerge, parsenode, nodeExprs, rawOptions},

    rawOptions = Join[
        KeyDrop[expr, {"hyperparameters", "dataflow", "options", "$schema"}],
        Lookup[expr, "options", <||>]
        ];

    topoptsrules = Normal @ KeyValueMap[
        Function[{k, v},
            Which[
                k === "multiplex" && ListQ[v], "multiplex" -> StringJoin[ToString /@ v],
                True, k -> v
            ]], rawOptions
        ];

    hyperparams = Lookup[expr, "hyperparameters", <||>];

    hprules = KeyValueMap[
        Function[{k, v},
            Which[
                k === "default", _ -> v,
                StringQ[k] && StringMatchQ[k, NumberString], ToExpression[k] -> v, 
                True, k -> v 
            ]],
        hyperparams
        ];
    
    buildpath[p_] := 
        Module[{slot, tags = "", tagstr = "", gatingstr = "", userlabel = ""},
        
        If[IntegerQ[p] || StringQ[p], Return[p]];
        slot = p["slot"];
        If[KeyExistsQ[p, "tags"], tagstr = StringJoin[Lookup[Circuit`TagMap, p["tags"], ""]]];
        If[KeyExistsQ[p, "gating"], gatingstr = "@" <> StringJoin[ToString /@ p["gating"]]];
        If[KeyExistsQ[p, "label"], userlabel = p["label"] <> " "];
        
        tags = userlabel <> tagstr <> gatingstr;
        If[tags === "", slot, slot * tags]
        ];

    buildmerge[l_List] := buildpath /@ l;
    buildmerge[p_] := buildpath[p];

    parsenode[node_Association] := Module[

        {comp,  plugin, opts, send, receive, compExpr},

        comp = node["component"]; 
        plugin = Lookup[node, "plugin", Null];
        opts = Normal[KeyDrop[node, {"component", "plugin", "send", "receive"}]];       
        compExpr = comp @@ DeleteCases[Join[If[plugin =!= Null, {plugin}, {}], opts], Null];

        send = Sequence @@ Map[buildmerge, Lookup[node, "send", {}]];

        receive = Sequence @@ Map[buildmerge, Lookup[node, "receive", {}]];

        compExpr[send][receive]
        ];

    dataflow = Lookup[expr, "dataflow", {}];
    nodeExprs = parsenode /@ dataflow;

    Join[nodeExprs, hprules, topoptsrules]
    ]

    {decodeBytes, cleanString, formatrootelement, rootStrList, dfStrList, dfStr, prepareNode},

        If[Max[ToCharacterCode[s]] <= 255,

            ExportString[

                "RawJSON", "Compact" -> True
            ];
    
    rootStrList = formatrootelement /@ Keys[KeyDrop[expr, {"dataflow"}]];

    prepareNode[node_Association] := Module[{n = node},
        If[KeyExistsQ[n, "component"], 
            n["component"] = ToString[n["component"]]];
            
        If[KeyExistsQ[n, "plugin"], 
            n["plugin"] = ToString[n["plugin"], InputForm]];
    
        n
        ];  
      
    dfStrList = (cleanString @ decodeBytes @ 
        ExportString[prepareNode[#], "RawJSON", "Compact" -> True] &) /@ 
            expr["dataflow"];

    dfStr = "\"dataflow\": [\n  " <> StringRiffle[dfStrList, ",\n\n "] 
            <> "\n  ]";

    "{\n  " <> StringRiffle[Append[rootStrList, dfStr], ",\n\n  "] <> "\n}"
    ]

Memory[config_Association] :=

    Module[ {mem = <||>, NA, PA, NB, PB, T = 1, capacity, zero,
        store, retrieve, clear, memorycount},
            
        {NA, PA} = config["A_parameters"];
        {NB, PB} = config["B_parameters"];
        
        If[ ! MatchQ[{NA, PA, NB, PB}, {__Integer}], 
                    Message[Memory::params, config]];   

        (* Memory capacity, needed for default threshold calculation. *)
        capacity = If[NA == NB && PA == PB, 
                Log[2.0] NA (NA - 1) (NA - 2) / (PA (PA - 1) (PA - 2)),
                Log[2.0] NA (NA - 1) NB / (PA (PA - 1) PB)];
            
        (* Default pattern matching threshold. *)
        While[T < PA && capacity^2 * Binomial[PA, T] * Binomial[NA-PA, PA-T] / 
                Binomial[NA, PA] >= 1, T++];

        (* User-defined (scaled) threshold . *)
        If[KeyExistsQ[config, "threshold"] && NumberQ[config["threshold"]],
            T = Round[config["threshold"] * PA]];
            
        If[T < 2, T = 2];

        zero = Developer`ToPackedArray[ConstantArray[0, NB]];

        (* Store hetero-association A -> B in memory. *)
        store[A_List] := store[A, A];
        
        (* Store A -> B in memory. *)
        store[A_List, B_List] := Module[{v, sub2},
            (* Step 1: Expansion coding. *)
            sub2 = Subsets[A, {2}];
        
            (* Step 2: Memory update. *)
            v = zero; v[[B]] = 1; 
            (mem[#] = BitOr[Lookup[mem, Key[#], zero], v])& /@ sub2; 
            ];

        (* Memory retrieval. *)
        retrieve[A_List] := 
            Module[ {X, P, Y, R, Ri, sub2, v, t, w, ws, h, cutoff},
            (* Step 1: Initialize. *)
            X = A; 
            
            While[ True,
                (* Step 2: Threshold check.  *)
                If[ (P = Length[X]) < T, Return[{}]]; 
            
                (* Step 3: Expansion coding. *)
                sub2 = Subsets[Range[P], {2}];
            
                (* Step 4: Aggregate. *)
                Ri = ConstantArray[0, {P, NB}];
                (v = Lookup[mem, Key[{X[[#1]], X[[#2]]}], zero]; 
                        Ri[[#1]] += v; Ri[[#2]] += v) & @@@ sub2;
                R = (Plus @@ Ri) / 2; 
            
                (* Step 5: Select (kWTA with threshold T(T-1)/2) *)
                (* The first element of a negative Ordering is the index 
                    of the k-th largest value. 
                    Use this for the threshold check. *)
                t = Max[1, R[[First @ Ordering[R, -PB]]]];
                
                (* Step 6: Threshold check. *)
                If[ t < T (T - 1) / 2, Return[{}]]; 
                Y = Sort[Pick[Range[NB], UnitStep[R - t], 1]];
            
                (* Step 7: Per-element weights. *)
                w = Total /@ Ri[[All, Y]];
                
                (* Step 8: Convergence test.  *)
                ws = Sort[w]; 
                h = P;
                While[ (cutoff = ws[[-h]]) < Length[Y] * (h - 1), --h];

                If[ h < T, Return[{}]];         
                
                (* Step 9: Refine. *)
                X = Pick[X, # >= cutoff & /@ w];
                
                (* Handle edge case. *)
                If[h === T &&  h < Length[X], X = {}; Return[{}]];
                
                (* Finished if X has converged. *)
                If[Length[X] == P, Return[Y]]; 

                (* Step 10: Iterate.  *)
                ]
            ];
            
        clear := (mem = <||>;);
        memorycount := Total[Values[mem], 2];
                
        <|
            "A_parameters" -> {NA, PA},
            "B_parameters" -> {NB, PB},
            "T" -> T, (* Absolute pattern matching threshold. *)
            "store" -> store,
            "retrieve" -> retrieve,
            "clear" :> clear,
            "memorycount" :> memorycount,
            "backend" -> "Reference"
        |>
        ]

Memory[config_Association] :=

    Module[ {mem = <||>, NA, PA, NB, PB, T = 1, capacity, zero,
        store, retrieve, clear, memorycount, accumulate},
            
        {NA, PA} = config["A_parameters"];
        {NB, PB} = config["B_parameters"];
        
        If[ ! MatchQ[{NA, PA, NB, PB}, {__Integer}], 
                    Message[Memory::params, config]];   

        (* Memory capacity, needed for default threshold calculation. *)
        capacity = If[NA == NB && PA == PB, 
                Log[2.0] NA (NA - 1) (NA - 2) / (PA (PA - 1) (PA - 2)),
                Log[2.0] NA (NA - 1) NB / (PA (PA - 1) PB)];
            
        (* Default pattern matching threshold. *)
        While[T < PA && capacity^2 * Binomial[PA, T] * Binomial[NA-PA, PA-T] / 
                Binomial[NA, PA] >= 1, T++];

        (* User-defined (scaled) threshold . *)
        If[KeyExistsQ[config, "threshold"] && NumberQ[config["threshold"]],
            T = Round[config["threshold"] * PA]];
            
        If[T < 2, T = 2];

        zero = Developer`ToPackedArray[ConstantArray[0, NB]];

        (* Store auto-association A -> A in memory. *)
        store[A_List] := store[A, A];
        
        (* Store hetero-association A -> B in memory. *)
        store[A_List, B_List] := Module[{keys, existing},
            If[Length[A] < 2, Return[]];
            
            (* Step 1: Expansion coding. *)
            keys = Subsets[A, {2}];
            
            (* Step 2: Bulk memory retrieval. *)
            existing = Lookup[mem, keys, zero];
            
            (* Step 3: Vectorized memory update. *)
            existing[[All, B]] = 1;
            
            (* Step 4: Bulk memory commit. *)
            AssociateTo[mem, AssociationThread[keys, existing]];
        ];
        
        (* Helper function. *)
        accumulate = Compile[{{P, _Integer}, {valsY, _Integer, 1}},
            Module[{w = Table[0, {P}], k = 1},
                Do[
                    w[[i]] += valsY[[k]];
                    w[[j]] += valsY[[k]];
                    k++,
                {i, 1, P - 1}, {j, i + 1, P} ];
                w], RuntimeOptions -> "Speed"
            ];

        (* Memory retrieval. *)
        retrieve[A_List] := 
            Module[{X = A, P, Y, R, vals, valsY, t, w, ws, h, cutoff},
            
            While[True,
            
                P = Length[X];
                If[P < T, Return[{}]];

                (* Bulk native lookup returning a 2D packed C-array *)
                vals = Lookup[mem, Subsets[X, {2}], zero];

                (* Vectorized column accumulation *)
                R = Total[vals];

                t = Max[1, R[[First @ Ordering[R, -PB]]]];
                If[t < T (T - 1) / 2, Return[{}]];

                Y = Pick[Range[NB], UnitStep[R - t], 1];

                (* Slice array to columns Y and row-reduce to scalar pair weights *)
                valsY = Total[vals[[All, Y]], {2}];

                (* 1D compiled distribution to element weights *)
                w = accumulate[P, valsY];

                ws = Sort[w];
                h = P;
                While[(cutoff = ws[[-h]]) < Length[Y] * (h - 1), --h];

                If[h < T, Return[{}]];

                X = Pick[X, UnitStep[w - cutoff], 1];

                If[h === T && h < Length[X], Return[{}]];
                If[Length[X] == P, Return[Y]];
                ]
            ];  
        
        clear := (mem = <||>;);
        memorycount := Total[Values[mem], 2];
                
        <|
            "A_parameters" -> {NA, PA},
            "B_parameters" -> {NB, PB},
            "T" -> T, (* Absolute pattern matching threshold. *)
            "store" -> store,
            "retrieve" -> retrieve,
            "clear" :> clear,
            "memorycount" :> memorycount,
            "backend" -> "Mathematica"
        |>
        ]

(*
Uses the Standard C backend implementation of topological associative memory.
*)


Memory[config_Association] :=

    Module[ {M, NA, PA, NB, PB, T, new, setthreshold, getthreshold,
               memory, read, write, store, retrieve, free, clear, memorycount},

        new = ForeignFunctionLoad["TAM", "TAMnew", 
            {"CInt", "CInt", "CInt", "CInt"} -> "OpaqueRawPointer"];

        setthreshold = ForeignFunctionLoad["TAM", "TAMsetthreshold", 
            {"OpaqueRawPointer", "CInt"} -> "Void"];                

        getthreshold = ForeignFunctionLoad["TAM", "TAMgetthreshold", 
            {"OpaqueRawPointer"} -> "CInt"];

        memory = ForeignFunctionLoad["TAM", "TAMmemory", 
            {"OpaqueRawPointer"} -> "CInt"];

        write = ForeignFunctionLoad["TAM", "TAMwrite", 
            {"OpaqueRawPointer", "RawPointer"::["CInt"], "CInt", 
            "RawPointer"::["CInt"], "CInt"} -> "Void"];

        read = ForeignFunctionLoad["TAM", "TAMread", 
            {"OpaqueRawPointer", "RawPointer"::["CInt"], "CInt", 
            "RawPointer"::["CInt"], "RawPointer"::["CInt"]} -> "Void"];

        free = ForeignFunctionLoad["TAM", "TAMfree", 
            {"OpaqueRawPointer"} -> "Void"];
    
        {NA, PA} = config["A_parameters"];
        {NB, PB} = config["B_parameters"];
        
        If[ ! MatchQ[{NA, PA, NB, PB}, {__Integer}], 
                    Message[Memory::params, config]];   

        M = new[NA, PA, NB, PB]; (* Memory constructor. *)

        (* Push user-defined (scaled) threshold. *)
        If[KeyExistsQ[config, "threshold"] && NumberQ[config["threshold"]],
            setthreshold[M, Round[config["threshold"] * PA]]];

        (* Get final threshold. *)
        T = getthreshold[M];
                                                                
        (* Store auto-association A -> A in memory. *)
        store[A_List] := store[A, A];

        (* Store hetero-association A -> B in memory. *)
        store[A_List, B_List] := 
            If[Length[A] > 0 && Length[B] > 0, 
                write[M, RawMemoryExport[A, "CInt"], 
                Length[A], RawMemoryExport[B, "CInt"], Length[B]]];
        
        (* Retrieve. *)
        retrieve[A_List] := Module[ {Y, Yn},
            If[Length[A] === 0, Return[{}]];
            Y = RawMemoryAllocate["CInt", NB];
            Yn = RawMemoryAllocate["CInt"];
            read[M, RawMemoryExport[A, "CInt"], Length[A], Y, Yn];
            RawMemoryImport[Y, {"List", RawMemoryRead[Yn]}]
            ];
        
        clear := free[M];
        memorycount := memory[M];
        
        <|
            "A_parameters" -> {NA, PA},
            "B_parameters" -> {NB, PB},
            "T" -> T, (* Absolute pattern matching threshold. *)
            "store" -> store,
            "retrieve" -> retrieve,
            "clear" :> clear,
            "memorycount" :> memorycount,
            "backend" -> "C_FFI"
        |>
        ]
